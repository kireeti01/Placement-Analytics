const confirmResetPassword = async () => {
  try {
    console.log('🔑 Resetting password for admin:', resetPasswordData.adminId);
    console.log('📝 New password:', resetPasswordData.newPassword);
    
    const response = await collegeAPI.resetAdminPassword(
      resetPasswordData.adminId, 
      resetPasswordData.newPassword
    );
    
    console.log('✅ Reset password response:', response);
    
    toast.success('✅ Password reset successfully!');
    toast.info('New password: ' + resetPasswordData.newPassword, { duration: 10000 });
    setShowResetPassword(false);
    loadData();
  } catch (error) {
    console.error('❌ Reset password error:', error);
    console.error('❌ Error response:', error.response?.data);
    
    if (error.response && error.response.status === 200) {
      toast.success('✅ Password reset successfully!');
      toast.info('New password: ' + resetPasswordData.newPassword, { duration: 10000 });
      setShowResetPassword(false);
      loadData();
    } else {
      const errorMsg = error.response?.data?.error || 'Unknown error';
      toast.error('❌ Failed to reset password: ' + errorMsg);
    }
  }
};
